### 1.1 CONNECT ###############################################################
# RDP into the VDI from your laptop:

# (GUI alternative: remmina)

# SSH to a Unix target - direct, or jump through the VDI:
ssh <user>@<target>
ssh -p <port> <user>@<target>              # non-default port (check nmap)
ssh -i key.pem <user>@<target>             # key auth (chmod 600 key.pem first)
ssh -J <user>@<VDI> <user>@<target>        # ProxyJump through the VDI

# Reusable ~/.ssh/config (type a name instead of the whole hop):
#   Host vdi
#       HostName <VDI-IP>
#       User <user>
#       ServerAliveInterval 60
#   Host farm-*
#       ProxyJump vdi
#       User <user>
#   -> then:  ssh farm-web   ssh farm-db   ...

# Pivoting to reach hosts not directly routable:
ssh -L 8080:<internal-host>:80 <user>@<VDI>    # local fwd -> http://localhost:8080
ssh -L 13306:<db-host>:3306 <user>@<VDI>       # reach an internal DB locally
ssh -D 1080 <user>@<VDI>                        # dynamic SOCKS proxy
#   then add to /etc/proxychains4.conf:  socks5 127.0.0.1 1080
#   and run tools as:  proxychains4 <tool> ...
sshuttle -r <user>@<VDI> <subnet>/24            # transparent whole-subnet pivot

### 1.5 PER-SERVICE DETAIL (enumerate -> verify, all non-destructive) #########

# --- FTP 21 ---
nmap -Pn -p21 -sV --script ftp-anon,ftp-syst,ftp-bounce $IP
ftp $IP          # login: anonymous / <blank or any email>
#   FINDING: anon access, cleartext creds, writable root. Proof: list + grab 1 file.

# --- SSH 22 ---
nmap -Pn -p22 -sV --script ssh2-enum-algos,ssh-auth-methods,ssh-hostkey $IP
#   FINDING: weak KEX/cipher, password auth + weak creds (lockout-aware), old ver.

# --- TELNET 23 ---
nmap -Pn -p23 -sV --script telnet-encryption $IP
#   FINDING: cleartext admin protocol (finding by existence), default creds.

# --- SMTP 25/587 ---
nmap -Pn -p25,587 -sV --script smtp-commands,smtp-open-relay,smtp-enum-users $IP
smtp-user-enum -M VRFY -U users.txt -t $IP        # a few names, not a dump
#   FINDING: open relay (High), VRFY user enum, no TLS.

# --- DNS 53 ---
dig axfr @$IP $DOMAIN                               # zone transfer = full map
dig +short version.bind chaos txt @$IP
nmap -Pn -p53 --script dns-nsid,dns-recursion $IP
#   FINDING: AXFR (Med-High), open recursion.

# --- POP3/IMAP 110/143/993/995 ---
nmap -Pn -p110,143,993,995 -sV --script pop3-capabilities,imap-capabilities $IP
#   FINDING: cleartext mail auth (no STARTTLS).

# --- SMB 139/445 ---
nxc smb $IP -u '' -p ''                             # null session
nxc smb $IP -u 'guest' -p '' --shares               # guest + shares
nxc smb $IP -u '' -p '' --shares
smbclient -L //$IP/ -N
smbclient //$IP/<share> -N                          # read one file as proof
enum4linux-ng -A $IP
nmap -Pn -p445 --script smb2-security-mode,smb-enum-shares,smb-os-discovery $IP
nxc smb $IP --gen-relay-list relay.txt              # hosts w/ signing OFF (relay)
#   FINDING: anon/guest share (High), signing disabled (relay enabler), secrets in shares.

# --- SNMP 161/udp ---
onesixtyone -c /usr/share/seclists/Discovery/SNMP/common-snmp-community-strings.txt $IP
snmpwalk -v2c -c public $IP
snmp-check $IP -c public
#   FINDING: default community -> config + sometimes creds (Med-High).

# --- LDAP 389/636 ---
ldapsearch -x -H ldap://$IP -s base namingContexts
ldapsearch -x -H ldap://$IP -b "dc=butters,dc=local"
nxc ldap $IP -u '' -p ''
#   FINDING: anon bind -> user enum, secrets in description/info fields.

# --- NFS 111/2049 ---
showmount -e $IP
sudo mkdir -p /mnt/nfs && sudo mount -t nfs -o ro,nolock $IP:/<export> /mnt/nfs
#   FINDING: world-readable export (High), no_root_squash (priv-esc path).

# --- rsync 873 ---
rsync --contimeout=5 --list-only rsync://$IP/
#   FINDING: anonymous module access.

# --- IPMI 623/udp ---

#   FINDING: cipher-0 auth bypass / RAKP hash disclosure (High).

# --- VNC 5900 ---

#   FINDING: no-auth / weak VNC (High).

# --- RDP 3389 ---
